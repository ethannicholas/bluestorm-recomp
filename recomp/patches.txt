# The frame rate (runtime/pace.cpp): fn_80007A94 ends a frame by spinning until N retraces
# have passed since its mark (N = 1 in play, 2 in mode 0) and then waiting for one more;
# at 60 frames a second N is 0. The mode runner fn_80006D20 then advances the game's clock
# (0x806919F0) by the 1/30 at 0x80692400 and its frame counter (0x80690FF8) by one; at 60
# the clock advances by half and the counter every other frame.
80007B3C { uint32_t wr_fields_per_frame(uint32_t); c->r[31] = wr_fields_per_frame(1u); }
80007B44 { uint32_t wr_fields_per_frame(uint32_t); c->r[31] = wr_fields_per_frame(2u); }
80006F28 { double wr_frame_dt(double); double v = (double)(float)(c->f[0].d + wr_frame_dt(c->f[31].d)); c->f[0].d = v; c->ps1[0] = v; }
80006F14 { uint32_t wr_frame_count_step(uint32_t); c->r[0] = c->r[3] + wr_frame_count_step(c->r[3]); }
# The water (docs/dev/performance.md, fifth pass): fn_800BDDC4 is the water-and-effects
# update, once a frame: it pushes each rider's hull into the 64-by-64 field of integer
# heights (fn_8000E94C), runs the wave simulation on it (fn_80008FC4), then the keyframe
# animation player (fn_8000CCF0) and fn_8000E9B0. A wave equation cannot be halved store
# by store, and gating only the simulation left the pushes at twice the rate and the foam
# white, so at 60 frames a second the whole update runs on every other frame instead:
# gcn_step_int is 1 on the frames a step falls on and 0 between. fn_80013830 is the same
# simulation for the title screen's water.

800BBD20 { if (gcn_step_int(0x800BBD20u, 1)) { c->lr = 0x800BBD24u; fn_800BDDC4(c); } }
800141A0 { if (gcn_step_int(0x800141A0u, 1)) { c->lr = 0x800141A4u; fn_80013830(c); } }
800147A0 { if (gcn_step_int(0x800147A0u, 1)) { c->lr = 0x800147A4u; fn_80013830(c); } }
